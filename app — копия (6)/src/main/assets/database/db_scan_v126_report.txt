SQL: CREATE TABLE equipment_category_links (

    category_index TEXT NOT NULL,

SQL: CREATE TABLE weapons (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    index_name TEXT NOT NULL UNIQUE,
    name TEXT NOT NULL,
    category_index TEXT,
    damage_dice TEXT,
    damage_type TEXT,
    weight REAL,
    properties_json TEXT,
    rarity TEXT,
    cost_json TEXT,
    cost_cp INTEGER DEFAULT 0

    category_index TEXT,

    stealth_disadvantage INTEGER,
